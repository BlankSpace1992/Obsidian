## Git

```bash
# 分支
git branch -a                    # 查看所有分支（含远程）
git checkout -b <branch>         # 新建并切换分支
git branch -d <branch>           # 删除本地分支

# 提交
git add . && git commit -m ""    # 一步提交
git push origin HEAD:<branch>    # 推送到指定远程分支
git push --set-upstream origin <branch>  # 首次推送并关联

# 回退
git reset --soft HEAD~1          # 撤销 commit，保留改动
git checkout -- <file>           # 丢弃工作区改动
git reset HEAD <file>            # 取消暂存

# 日志
git log --oneline -10            # 最近 10 条简洁日志
git diff                         # 查看未暂存的改动
git diff --staged                # 查看已暂存的改动

# 暂存
git stash                        # 暂存当前改动
git stash pop                    # 恢复最近一次暂存
git stash list                   # 查看暂存列表
```

## Docker

```bash
# 容器
docker ps -a                     # 查看所有容器
docker start/stop/restart <id>   # 启动/停止/重启
docker rm <id>                   # 删除容器
docker logs -f --tail 100 <id>   # 查看日志

# 镜像
docker images                    # 查看本地镜像
docker pull <image:tag>          # 拉取镜像
docker rmi <image>               # 删除镜像

# Compose
docker-compose up -d             # 后台启动
docker-compose down              # 停止并删除
docker-compose restart           # 重启服务
docker-compose logs -f           # 查看日志
```

## Linux 常用

```bash
# 文件
ls -lh                           # 人类可读的文件列表
du -sh * | sort -rh | head -10   # 当前目录下最大的 10 个文件/夹
truncate -s 0 nohup.out          # 清空 nohup.out（不删文件，释放磁盘）
find . -name "*.log" -mtime +7 -delete  # 删除 7 天前的日志
find / -type f -exec du -h {} + | sort -hr | head -n 10  # 找出系统最大的 10 个文件

# 进程
ps aux | grep <关键词>
kill -9 <PID>
lsof -i :<端口>                  # 查看端口占用

# 磁盘
df -h                            # 磁盘使用情况
free -h                          # 内存使用情况

# 网络
curl -I <url>                    # 只看响应头
ss -tlnp                         # 查看监听端口
```
